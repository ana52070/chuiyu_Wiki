相对比于c语言：无需使用typedef

```cpp

#include <iostream>

using namespace std;

struct stu
{
    string name;
    int age;
};


int main()
{
    // c语言表示方法:
    struct stu xiaoming;
    xiaoming.age = 1;


    //cpp省略写法:
    stu zhangsan;
    zhangsan.age = 1;

    return 0;
}

```
